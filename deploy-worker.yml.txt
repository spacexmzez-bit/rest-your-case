# File name: .github/workflows/deploy-worker.yml
name: Deploy RYC Worker

on:
  push:
    branches: [main]
    paths:
      - worker.js
      - standard-support.js
      - game-state.js
      - data.js
      - game.js
      - game-scene.js
      - cases.js
      - scout_prompt.js
      - tests/standard-update.test.mjs
      - wrangler.toml
      - .github/workflows/deploy-worker.yml
  workflow_dispatch:

permissions:
  contents: read

concurrency:
  group: ryc-worker-production
  cancel-in-progress: false

jobs:
  deploy:
    runs-on: ubuntu-latest
    timeout-minutes: 10
    steps:
      - name: Download repository files
        uses: actions/checkout@v6

      - name: Set up Node
        uses: actions/setup-node@v7
        with:
          node-version: "24"
          package-manager-cache: false

      - name: Check Standard scripts
        run: |
          for file in standard-support.js game-state.js data.js game.js game-scene.js cases.js scout_prompt.js; do
            node --check "$file"
          done
          node --test tests/standard-update.test.mjs

      - name: Check Worker syntax
        run: node --input-type=module --check < worker.js

      - name: Check deployment credentials
        env:
          CLOUDFLARE_API_TOKEN: ${{ secrets.CLOUDFLARE_API_TOKEN }}
          CLOUDFLARE_ACCOUNT_ID: ${{ secrets.CLOUDFLARE_ACCOUNT_ID }}
        run: |
          if [ -z "$CLOUDFLARE_API_TOKEN" ]; then
            echo "::error::Add CLOUDFLARE_API_TOKEN in repository Settings > Secrets and variables > Actions."
            exit 1
          fi
          if [ -z "$CLOUDFLARE_ACCOUNT_ID" ]; then
            echo "::error::Add CLOUDFLARE_ACCOUNT_ID in repository Settings > Secrets and variables > Actions."
            exit 1
          fi

      - name: Deploy to existing Cloudflare Worker
        uses: cloudflare/wrangler-action@v4
        with:
          apiToken: ${{ secrets.CLOUDFLARE_API_TOKEN }}
          accountId: ${{ secrets.CLOUDFLARE_ACCOUNT_ID }}
          wranglerVersion: "4"

          command: deploy --config wrangler.toml
