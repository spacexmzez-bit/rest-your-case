  const text = (value, max = 4000) => typeof value === 'string' && value.trim() && value.length <= max;
  const statuses = ['Marked', 'Admitted', 'SUPPRESSED', 'Excluded'];
  function checkpoint(raw) {
    const matches = [...String(raw).matchAll(/<!--STATE_CHECKPOINT:\s*([\s\S]*?)-->/g)];
    if (matches.length > 1) error('Only one state checkpoint is allowed.');
    if (!matches.length) return null;
    try { const result = JSON.parse(matches[0][1]); if (!object(result)) error('Checkpoint must be an object.'); return result; }
    catch { error('Invalid state checkpoint JSON.'); }
  }
  function evidence(value) {
    if (!object(value) || !key(value.id || value.tag) || !text(value.name || value.title, 300) ||
      !text(value.details || value.facts || value.description, 8000)) error('Discovery requires an exhibit ID, title and complete description.');
    if (value.status !== undefined && !statuses.includes(value.status)) error('Unknown exhibit status.');
    return { ...value, id: value.id || value.tag, name: value.name || value.title,
      details: value.details || value.facts || value.description, status: value.status || 'Marked' };
  }
  function mergeDocket(old, incoming, channel) {
    const map = new Map(old.map(e => [String(e.id || e.tag), clone(e)]));
    for (const source of incoming) {
      if(!object(source) || !key(source.id || source.tag))error('Invalid exhibit identifier.');
      const exhibitId=source.id || source.tag, prior=map.get(exhibitId);
      if (prior?.isManual) {
        if(['name','title','details','facts','description','type','status'].some(k=>source[k]!==undefined && source[k]!==prior[k]))error('The engine cannot overwrite a manually entered exhibit.');
        continue;
      }
      // Select aliases from this update before consulting older fields. Otherwise an old
      // `details` string silently overwrites a newer `facts` string (likewise name/title).
      const titleKey=['name','title'].find(k=>source[k]!==undefined);
      const detailKey=['details','facts','description'].find(k=>source[k]!==undefined);
      if(titleKey && !text(source[titleKey],300))error('Invalid exhibit title.');
      if(detailKey && !text(source[detailKey],8000))error('Invalid exhibit description.');
      if(source.status!==undefined && !statuses.includes(source.status))error('Unknown exhibit status.');
      let item={...prior,...source,id:exhibitId,isApi:true,isManual:false};
      if(titleKey){item.name=source[titleKey];if('title' in item)item.title=item.name;}
      if(detailKey){item.details=source[detailKey];item.facts=item.details;if('description' in item)item.description=item.details;}
      // New evidence requires complete content. A legacy row may have no description;
      // a status-only update must preserve that row without inventing missing content.
      if(!prior)item=evidence(item);
      item.status=channel==='diaz' ? prior?.status || 'Marked' : source.status ?? prior?.status ?? 'Marked';
      if(item.details && !item.facts)item.facts=item.details;
      map.set(exhibitId,item);
    }
    return [...map.values()];
  }
  function validateDiscovery(raw, state) {
    const u = checkpoint(raw);
    if (!u || !object(u.investigation)) error('Diaz replies require an explicit investigation record, including advice-only replies. Retry; no resources were changed.');
    const report = u.investigation;
    if (!key(report.id) || !['advice', 'completed', 'not_completed'].includes(report.status) ||
      !Number.isInteger(report.cost) || report.cost < 0 || report.cost > 2 || !text(report.summary) || !text(report.insight)) error('Invalid investigation report.');
    if (report.status !== 'completed' && report.cost !== 0) error('Advice and unfinished investigations cost 0 AP.');
    if (u.ap !== state.ap - report.cost || u.ap < 0) error('Investigation AP must equal current AP minus the reported cost.');
    const exhibits = u.docket || [], facts = u.facts || [];
    if (!Array.isArray(exhibits) || !Array.isArray(facts) || facts.some(f => !text(f, 4000))) error('Malformed investigation result.');
    exhibits.forEach(evidence);
    if(report.status==='completed' && report.cost===0)error('A completed discovery requires its paid investigation cost; public-record advice costs zero.');
    if (report.status === 'completed' && !exhibits.length && !facts.length) error('Completed work must save an exhibit or a factual result, including negative findings.');
    if (report.status !== 'completed' && (exhibits.length || facts.length)) error('Advice/unfinished work cannot silently create discoveries.');
    const dialogue = raw.match(/<DIALOGUE\s*>([\s\S]*?)<\/DIALOGUE\s*>/i)?.[1] || '';
    const spoken = dialogue.replace(/<[^>]+>/g, '').replace(/&(?:amp|lt|gt|quot|apos);/g,m=>({'&amp;':'&','&lt;':'<','&gt;':'>','&quot;':'"','&apos;':"'"})[m]).replace(/&#(x[0-9a-f]+|[0-9]+);/gi,(_,n)=>String.fromCodePoint(n[0].toLowerCase()==='x'?parseInt(n.slice(1),16):Number(n)));
    if (!spoken.includes(report.summary) || !spoken.includes(report.insight)) error('Diaz must speak the saved result summary and the separate investigative insight.');
    if (exhibits.some(e => [e.details, e.facts, e.description].includes(report.insight))) error('Diaz insight must add context beyond the docket description.');
    return report;
  }
  function normalize(saved) {
    return { investigations: Array.isArray(saved?.investigations) ? saved.investigations.filter(object) : [],
      characters: Array.isArray(saved?.characters) ? saved.characters.filter(p => object(p) && text(p.name, 100) && ['F', 'M'].includes(p.gender)).map(p => ({ name: p.name, gender: p.gender })) : [],

      disclosures: Array.isArray(saved?.disclosures) ? saved.disclosures.filter(o => object(o) && key(o.id)) : [],
      clerkMemory: object(saved?.clerkMemory) ? saved.clerkMemory : {},
      caseOutcome: object(saved?.caseOutcome) ? saved.caseOutcome : null };
  }

          reducedCharge: offer.reducedCharge || '', sourceExhibitIds: offer.sourceExhibitIds,

    // any remaining proposals. The UI and validator now use the same boundary.
    for(const offer of next.offers)if(offer.status==='proposed' && next.turn>=offer.expiresTurn)offer.status='expired';
    return next;
  }

  function dialogueText(raw) {

    const block = String(raw).match(/<DIALOGUE\s*>([\s\S]*?)<\/DIALOGUE\s*>/i)?.[1];
    return (block || String(raw).replace(/<!--[\s\S]*?-->/g, ''))
      .replace(/={3,}\s*SEALED CASE GROUND TRUTH\s*={3,}[\s\S]*?(?:={3,}|$)/gi, '')
      .replace(/<SEALED_TRUTH\b[^>]*>[\s\S]*?<\/SEALED_TRUTH>/gi, '')
      .replace(/\[ENGINE INTERNAL[^\]]*\][\s\S]*$/gi, '')
      .replace(/^\s*(?:#{1,6}\s*)?(?:SEALED (?:TRUTH|RECORD)|GROUND TRUTH)\b[\s\S]*$/gmi, '')
      .replace(/<[^>]+>/g, '').trim();
  }
  function historySources(hist, channel) {

  }
  function memoryPacket(state, histories, channel, message) {
    const own = historySources(histories[channel] || [], channel);
    const court = historySources(histories.court || [], 'court');
    const terms = [...new Set(message.toLowerCase().match(/[\p{L}\p{N}]{3,}/gu) || [])].filter(t => !['the','and','that','this','with','what','from','you','can'].includes(t));
    const older = own.slice(0, -8);
    const relevant = [...court, ...older].map((e,i) => ({ e, i, score: terms.filter(t => e.text.toLowerCase().includes(t)).length + (message.includes(e.id) ? 100 : 0) }))
      .filter(x => x.score > 0).sort((a,b) => b.score-a.score || b.i-a.i).slice(0, 10).map(x=>x.e);
    // The clerk is an index of originals, not another AI call. No invented summaries or extra tokens.
    const corrections = older.filter(e=>/\b(?:correction|actually|instead|promis|agreed|not true|mistake)\b|تصحيح|بل |وعد|خطأ/i.test(e.text)).slice(-4);
    const sources = new Map([...court.slice(-4), ...relevant, ...corrections].map(e=>[e.id,e]));
    return { history: (histories[channel] || []).slice(-8), originals: [...sources.values()],
      sourceBoundary: 'Original attributed statements; a claim is not proof. Latest corrections govern. Private consultation channels stay separate. Ask for clarification when the supplied record does not establish an answer.' };
  }
  function publicCase(state) {

    return result;
  }

    }
    last.output+=last.thinking;
    for(const field of ['input','output','thinking','cached','repeated','unknown'])meter.totals[field]+=last[field];
    meter.last=last;return meter;

/**
 * Procedurally samples 1 seed per category matching the active complexity and category
 * @param {number} complexity - Level 1 to 5
 * @param {string} categoryKey - 'Violent Felony', 'White-Collar Fraud', 'Cybercrime / Espionage', or 'Random Case File'
 * @returns {object} Procedural case docket ready for Turn 1 injection
 */
function sampleCaseDocket(complexity = 3, categoryKey = "Random Case File") {
  const comp = Math.min(5, Math.max(1, parseInt(complexity, 10) || 3));

  // Helper to filter an array by [min, max] complexity range
  function filterByComplexity(arr, key = "complexity_range") {
    const valid = arr.filter(item => {
      const range = item[key] || item.range || [1, 5];
      return comp >= range[0] && comp <= range[1];
    });
    return valid.length > 0 ? valid : arr;
  }

  // Helper to pick 1 random element from an array
  function pickRandom(arr) {
    if (!arr || arr.length === 0) return null;
    return arr[Math.floor(Math.random() * arr.length)];
  }

  // 1. Resolve Crime Category Key to Internal Genre Key
  let genreKey = "violent_crimes";
  const catNormalized = String(categoryKey).toLowerCase();

  if (catNormalized.includes("white-collar") || catNormalized.includes("fraud")) {
    genreKey = "financial_fraud";
  } else if (catNormalized.includes("cyber") || catNormalized.includes("espionage")) {
    genreKey = "cyber_crime";
  } else if (catNormalized.includes("substance") || catNormalized.includes("narcotics")) {
    genreKey = "substance_use_and_distribution";
  } else if (catNormalized.includes("overcharge") || catNormalized.includes("manslaughter")) {
    genreKey = "overcharge_and_manslaughter";
  } else if (catNormalized.includes("random")) {
    const allGenreKeys = Object.keys(GAME_DATA.crime_genres);
    genreKey = pickRandom(allGenreKeys);
  }

  const selectedGenre = GAME_DATA.crime_genres[genreKey];
  const eligibleCharges = filterByComplexity(selectedGenre.charges);
  const pickedCharge = pickRandom(eligibleCharges);

  // 2. Sample Courtroom Roster
  const pickedJudge = pickRandom(GAME_DATA.judges);
  const pickedDA = pickRandom(GAME_DATA.district_attorneys);

  // 3. Sample Demographic Pool to generate a client name
  const ethnicityKeys = Object.keys(GAME_DATA.names_by_ethnicity);
  const chosenEthnicity = pickRandom(ethnicityKeys);
  const ethPool = GAME_DATA.names_by_ethnicity[chosenEthnicity];
  const isFemale = Math.random() > 0.5;
  const firstName = isFemale ? pickRandom(ethPool.first_names.female) : pickRandom(ethPool.first_names.male);
  const lastName = pickRandom(ethPool.last_names);
  const clientName = `${firstName} ${lastName}`;

  const usedNames = new Set([clientName, pickedJudge.name, pickedDA.name]);
  function samplePerson(id, role) {
    for (let attempt=0;attempt<100;attempt++) {
      const ethnicity=pickRandom(ethnicityKeys), pool=GAME_DATA.names_by_ethnicity[ethnicity];
      const gender=Math.random()>0.5?'F':'M';
      const name=pickRandom(pool.first_names[gender==='F'?'female':'male'])+' '+pickRandom(pool.last_names);
      if(!usedNames.has(name)){usedNames.add(name);return {id,name,gender,role,ethnicity};}
    }
    throw new Error('Could not sample distinct character names. Retry generation.');
  }
  const directViolence = /murder|manslaughter|assault|battery|homicide|robbery|kidnap|sexual|rape|death|false imprisonment|arson|carjacking|domestic violence/i.test(pickedCharge.charge);
  const victim=directViolence?samplePerson('VICTIM','victim'):null;
  const witnesses=Array.from({length:Math.min(6,2+comp)},(_,i)=>samplePerson('W'+String(i+1).padStart(2,'0'),'witness'));

  // 4. Sample Narrative & Evidentiary Seeds
  const pickedVenue = pickRandom(filterByComplexity(GAME_DATA.venues));
  const pickedAnchor = pickRandom(filterByComplexity(GAME_DATA.evidence_anchors));
  const pickedLegalFlaw = pickRandom(filterByComplexity(GAME_DATA.legal_flaws));
  const pickedWitnessFriction = pickRandom(filterByComplexity(GAME_DATA.witness_frictions));
  const pickedClientComplication = pickRandom(filterByComplexity(GAME_DATA.client_complications));

  // 5. Sample DA Misconduct Trigger & Verdict Guide
  const eligibleMisconduct = filterByComplexity(GAME_DATA.da_misconduct);
  const pickedMisconduct = pickRandom(eligibleMisconduct);
  const verdictGuide = GAME_DATA.jury_verdict_guides[genreKey];

  return {
    meta: {
      complexity: comp,
      genre_key: genreKey,
      genre_name: selectedGenre.genre
    },
    roster: {
      judge: pickedJudge,
      da: pickedDA,
      client: {
        name: clientName,
        gender: isFemale ? "F" : "M",
        ethnicity: chosenEthnicity,
        occupation: pickRandom(GAME_DATA.client_occupations)
      },
      victim, witnesses,
      investigator: "Investigator Carlos Diaz"
    },
    charge: pickedCharge,
    narrative_seeds: {
      venue: pickedVenue,
      primary_evidence_anchor: pickedAnchor,
      constitutional_flaw: pickedLegalFlaw,
      witness_friction: pickedWitnessFriction,
      client_complication: pickedClientComplication
    },
    mechanics: {
      da_misconduct_trap: pickedMisconduct,
      jury_verdict_guide: verdictGuide
    }
  };
}

    if (object(repaired.seed?.roster?.client)) {
      if (role) repaired.seed.roster.client.occupation = role;
      else delete repaired.seed.roster.client.occupation;

// This race bounds the whole operation, even if a transport does not honor abort.
// The operation receives a signal and cannot keep running a cascade after expiry.
async function bounded(operation, milliseconds, parentSignal, timeoutError) {
  const cancellation = () => typeof parentSignal?.reason?.code === 'string'
    ? parentSignal.reason : fault('REQUEST_CANCELLED', 'Request was cancelled.', 499);
  if (parentSignal?.aborted) throw cancellation();
  const controller = new AbortController();
  let timer, onAbort;
  const stop = new Promise((_, reject) => {
    onAbort = () => {
      const error = cancellation();
      controller.abort(error); reject(error);
    };
    if (parentSignal?.aborted) { onAbort(); return; }
    parentSignal?.addEventListener('abort', onAbort, {once: true});
    timer = setTimeout(() => {controller.abort(timeoutError); reject(timeoutError);}, milliseconds);
  });
  try {
    if (controller.signal.aborted) throw controller.signal.reason;
    return await Promise.race([Promise.resolve().then(() => operation(controller.signal)), stop]);
  } finally {
    clearTimeout(timer);
    parentSignal?.removeEventListener('abort', onAbort);
  }
}
function checkSignal(signal) {
  if (signal.aborted) throw signal.reason || fault('REQUEST_CANCELLED', 'Request was cancelled.', 499);
}

async function readRequest(request, signal) {
  const declaredLength = Number(request.headers.get('Content-Length'));
  if (declaredLength > MAX_REQUEST_BYTES) throw fault('PAYLOAD_TOO_LARGE', 'Payload exceeds 128 KiB.', 413);
  if (!request.body) throw fault('INVALID_JSON', 'Missing JSON request body.', 400);
  const reader = request.body.getReader(), chunks = [];

  const cancel = () => {reader.cancel().catch(() => {});};

  try {
    checkSignal(signal);
    while (true) {

      checkSignal(signal);

      if (!candidate) {lastError = fault(data?.promptFeedback?.blockReason ? 'AI_BLOCKED' : 'AI_EMPTY_RESPONSE', 'No usable model candidate was returned.'); continue;}
      if (candidate.finishReason && candidate.finishReason !== 'STOP') {
        lastError = fault(candidate.finishReason === 'MAX_TOKENS' ? 'AI_OUTPUT_TRUNCATED' : 'AI_BLOCKED', 'Model generation did not complete normally.'); continue;
      }
      const reply = Array.isArray(candidate.content?.parts) ? candidate.content.parts
        .filter(part => part?.thought !== true && typeof part?.text === 'string').map(part => part.text).join('') : '';
      if (!reply.trim()) {lastError = fault('AI_EMPTY_RESPONSE', 'Model returned no visible text.'); continue;}
      try {
        if (targetPersona === 'scout') validateScout(reply);
        else if (RYCDialogue.parse(reply, {strict: true}).length !== 1) throw fault('DIALOGUE_FORMAT_ERROR', 'Exactly one active speaker is required.');
      } catch (error) {lastError = fault(error.code || 'SCOUT_FORMAT_ERROR', error.message); continue;}
      if(body.contractVersion===2 && ['court','diaz'].includes(targetPersona)) {
        try {
          if(!body.caseState) throw new Error('Public case snapshot is missing.');
          const u=RYCStandard.checkpoint(reply), speaker=RYCDialogue.parse(reply,{strict:true})[0];
          if(u?.offers?.length && (speaker.role!=='prosecutor' || speaker.name!==body.caseState.profile.da)) throw new Error('Only the prosecutor may propose an offer.');
          if(u?.offerDecision?.decision==='accepted' && (speaker.role!=='judge' || speaker.name!==body.caseState.profile.judge)) throw new Error('Only the judge may approve an accepted resolution.');
          if(targetPersona==='diaz' && u?.investigation?.id!==body.operationId)throw new Error('Investigation ID must match the current operation ID.');
          RYCState.next(body.caseState,reply,targetPersona,{contractVersion:2,requestText:body.userText||'',undo:targetPersona==='court' && /^\/undo(?:\s|$)/i.test((body.userText||'').trim())});
        } catch(e) { lastError=fault('STATE_CONTRACT_ERROR',e.message); continue; }
      }
      checkSignal(signal);
      // Raw reply is retained for state extraction/history. Only tagged speech is displayed by the frontend.
      return jsonResponse({reply, activeModel: model, engineMode: targetPersona==='scout' ? 'light' : engineMode,usage:responseUsage()}, 200, cors);
    } catch (error) {
      checkSignal(signal);
      if(attempts.length===attemptStart)attempts.push({model,input:null,output:null,thinking:null,cached:null,total:null,status:null});
      if (error.code === 'AUTHENTICATION_ERROR') return jsonResponse({error:error.code,details:error.message,usage:responseUsage()},error.status||401,cors);
      lastError = error.code ? error : fault('UPSTREAM_NETWORK_ERROR', 'Unable to complete the upstream connection.');
    }
  }
  return jsonResponse({error:lastError.code,details:lastError.message,usage:responseUsage()},lastError.status||502,cors);
}
function validateScout(reply) {
  let value;
  try {value = JSON.parse(reply);} catch (_) {throw fault('SCOUT_FORMAT_ERROR', 'Scout returned invalid JSON.');}
  const nonempty = text => typeof text === 'string' && text.trim();
  if (!value || typeof value !== 'object' || !['caseTitle','crimeSummary','unencryptedTruth'].every(key => nonempty(value[key]))
      || !Array.isArray(value.starterExhibits) || value.starterExhibits.length !== 2) {
    throw fault('SCOUT_FORMAT_ERROR', 'Scout case is incomplete; exactly two starter exhibits are required.');
  }
  const ids = new Set();
  for (const exhibit of value.starterExhibits) {
    if (!exhibit || !['id','name','details'].every(key => nonempty(exhibit[key]))
        || !['Physical','Documentary','Digital','Forensic'].includes(exhibit.type) || exhibit.status !== 'Marked'
        || ids.has(exhibit.id)) throw fault('SCOUT_FORMAT_ERROR', 'Scout exhibits are invalid, duplicated or not Marked.');
    ids.add(exhibit.id);
  }
}
function formatGeminiContents(history, currentMessage) {
  const contents = [];
  for (const item of history) {
    if (!item || !Array.isArray(item.parts)) continue;
    const text = item.parts.map(part => typeof part === 'string' ? part : typeof part?.text === 'string' ? part.text : '')
      .filter(text => text.trim()).join('\n');
    if (!text) continue;
    const role = item.role === 'model' ? 'model' : 'user';
    if (!contents.length && role === 'model') continue;
    const previous = contents[contents.length - 1];
    // Preserve both consecutive turns rather than discarding the newer model response.
    if (previous?.role === role) previous.parts[0].text += '\n\n' + text;
    else contents.push({role, parts: [{text}]});
  }
  const last = contents[contents.length - 1];
  if (last?.role === 'user') last.parts[0].text += '\n\n' + currentMessage;
  else contents.push({role: 'user', parts: [{text: currentMessage}]});
  return contents;
}
